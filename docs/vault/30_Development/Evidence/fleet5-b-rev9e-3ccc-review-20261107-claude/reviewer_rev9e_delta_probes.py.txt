def unbound(source):
    tr = ast.parse(source); bound = set(dir(builtins))
    for n in ast.walk(tr):
        if isinstance(n, (ast.Import, ast.ImportFrom)): bound.update((a.asname or a.name).split('.')[0] for a in n.names)
        elif isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)): bound.add(n.name)
        elif isinstance(n, ast.arg): bound.add(n.arg)
        elif isinstance(n, ast.Name) and isinstance(n.ctx, (ast.Store, ast.Del)): bound.add(n.id)
        elif isinstance(n, ast.ExceptHandler) and n.name: bound.add(n.name)
        elif isinstance(n, (ast.MatchAs, ast.MatchStar)) and n.name: bound.add(n.name)
    return {n.id for n in ast.walk(tr) if isinstance(n, ast.Name) and isinstance(n.ctx, ast.Load) and n.id not in bound}

def witnesses(sources):
    checked = 0
    for w in W:
        src = sources.get(w['path']) or sb.frozen_source(w['path']); tree = ast.parse(src)
        node = next((n for n in ast.walk(tree) if hasattr(n, 'lineno') and [n.lineno, n.col_offset, n.end_lineno, n.end_col_offset] == w['span']), None)
        assert node is not None and sb.sha(ast.get_source_segment(src, node)) == w['expressionSHA256'], ('WITNESS', w)
        checked += 1
    return checked

def must_replace(src, old, new):
    assert src.count(old) == 1, ('ANCHOR', old[:80], src.count(old)); return src.replace(old, new, 1)

def roots_preserved(universe):
    census = cg.current_census(universe); facts = {p: sb.Facts(p, s) for p, s in universe.items() if p in S}
    keys = Counter(cg.root_key(r, facts[r['source']['path']]) for r in census['production'] if r['source']['path'] in facts)
    return census, not (FROZEN_ROOTS - keys)

def run(name, changes=None, world=None, expect=None, note=None, graph_only=False, finding=None):
    changes = changes or {}; world = world or {}
    changed = {**S, **changes}
    for p, s in {**changes, **world}.items():
        ast.parse(s)
        assert s.count('\n') >= sb.frozen_source(p).count('\n') or p in world, (name, 'LINE_LOSS', p)
        nb = sorted(unbound(s) - unbound(sb.frozen_source(p))); assert not nb, (name, nb)
    checked = witnesses({**world, **changed})
    c2 = {p: changed[p] for p in core}; k2 = {p: changed[p] for p in ctors}
    if graph_only: out, err = cg.prove({**c2, **k2}, world or None)
    else: out, err = t.provenance(c2, k2, universe_overrides=world or None)
    universe = {**world, **changed}
    census, preserved = roots_preserved(universe)
    disc, derr = fd.discover(census['production'], universe)
    audits = {a['path']: [h['kind'] for h in a['newRelevantHits']] for a in census['reflectionAudits'] + (disc['reflectionAudits'] if disc else [])}
    observed = 'ACCEPT' if err is None else 'REFUSE'
    row = {'case': name, 'observed': observed, 'reason': err, 'expected': expect,
           'matchesExpectation': expect is None or observed == expect.split(':', 1)[0] and (':' not in expect or (err or '').startswith(expect.split(':', 1)[1])),
           'finding': finding, 'mode': 'cg.prove graph only' if graph_only else 't1_provenance.provenance (full)',
           'astValid': True, 'newUnboundNames': [], 'witnessesChecked': checked,
           'mutatedSHA256': {p: sb.sha(s) for p, s in sorted({**changes, **world}.items())},
           'visibleProductionRoots': len(census['production']), 'sixFrozenRootsPreserved': preserved, 'discoveryError': derr,
           'newRelevantReflectionHitsByPath': {p: k for p, k in sorted(audits.items()) if k}, 'note': note}
    R.append(row); print(json.dumps({'case': name, 'observed': observed, 'reason': err, 'expected': expect}), flush=True)
    return row, out
