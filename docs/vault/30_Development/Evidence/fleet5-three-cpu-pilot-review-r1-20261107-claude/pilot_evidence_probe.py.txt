def tlv(b, i):
    t = b[i]; l = b[i + 1]; i += 2
    if l & 0x80:
        n = l & 0x7f; l = int.from_bytes(b[i:i + n], 'big'); i += n
    return t, b[i:i + l], i + l
def seq(b):
    r, i = [], 0
    while i < len(b):
        t, v, i = tlv(b, i); r.append((t, v))
    return r
def oid(v):
    a = [v[0] // 40, v[0] % 40]; n = 0
    for x in v[1:]:
        n = (n << 7) | (x & 0x7f)
        if not x & 0x80: a.append(n); n = 0
    return '.'.join(map(str, a))
def name_cn(v):
    for _, rdn in seq(v):
        for _, atv in seq(rdn):
            p = seq(atv)
            if oid(p[0][1]) == '2.5.4.3': return p[1][1].decode()
def gtime(t, v):
    s = v.decode(); fmt = '%y%m%d%H%M%SZ' if t == 0x17 else '%Y%m%d%H%M%SZ'
    return datetime.strptime(s, fmt).replace(tzinfo=timezone.utc)
def spki(v):
    p = seq(v); alg = oid(seq(p[0][1])[0][1]); return alg, p[1][1][1:]
def exts(v):
    r = {}
    for _, e in seq(v):
        p = seq(e); o = oid(p[0][1]); crit = len(p) == 3 and p[1][1] == b'\xff'; r[o] = (crit, p[-1][1])
    return r
def cert(der):
    top = seq(seq(der)[0][1]); tbs = seq(top[0][1]); i = 0
    if tbs[0][0] == 0xa0: i = 1
    serial, sigalg, issuer, validity, subject, pk = tbs[i:i + 6]
    ext = {}
    for t, v in tbs[i + 6:]:
        if t == 0xa3: ext = exts(seq(v)[0][1])
    nb, na = [gtime(t, v) for t, v in seq(validity[1])]
    d = {'issuerCN': name_cn(issuer[1]), 'subjectCN': name_cn(subject[1]), 'notBefore': nb.isoformat(), 'notAfter': na.isoformat(),
         'tbsSigAlg': oid(seq(sigalg[1])[0][1]), 'outerSigAlg': oid(seq(top[1][1])[0][1])}
    d['spkiAlg'], d['pub'] = spki(pk[1]); d['extOIDs'] = sorted(ext)
    if '2.5.29.19' in ext:
        bc = seq(seq(ext['2.5.29.19'][1])[0][1]) if ext['2.5.29.19'][1] != b'\x30\x00' else []
        d['basicConstraintsCritical'] = ext['2.5.29.19'][0]; d['CA'] = bool(bc and bc[0][0] == 0x01 and bc[0][1] == b'\xff')
    if '2.5.29.17' in ext:
        d['uris'] = [v.decode() for t, v in seq(seq(ext['2.5.29.17'][1])[0][1]) if t == 0x86]
        d['ips'] = ['.'.join(map(str, v)) for t, v in seq(seq(ext['2.5.29.17'][1])[0][1]) if t == 0x87]
    if '2.5.29.37' in ext: d['EKU'] = [oid(v) for _, v in seq(seq(ext['2.5.29.37'][1])[0][1])]
    if '2.5.29.14' in ext: d['SKI'] = seq(ext['2.5.29.14'][1])[0][1].hex()
    if '2.5.29.35' in ext: d['AKI'] = [v.hex() for t, v in seq(seq(ext['2.5.29.35'][1])[0][1]) if t == 0x80]
    return d, na
def csr(der):
    info = seq(seq(seq(der)[0][1])[0][1]); alg, pub = spki(info[2][1]); uris = []
    if len(info) > 3:
        for _, attr in seq(info[3][1]):
            p = seq(attr)
            if oid(p[0][1]) == '1.2.840.113549.1.9.14':
                for _, e in seq(seq(p[1][1])[0][1]):
                    q = seq(e)
                    if oid(q[0][1]) == '2.5.29.17': uris = [v.decode() for t, v in seq(seq(q[-1][1])[0][1]) if t == 0x86]
    return {'subjectCN': name_cn(info[1][1]), 'spkiAlg': alg, 'pub': pub, 'uris': uris}
