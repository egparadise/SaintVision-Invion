BASE, LEFT, RIGHT, LOWBASE = 'ff3169d360a707415d9b03484cf0cf3031f3912c', '21059451c23cebe8c038f76e437c390120205998', 'cc82fb80e5913a739c9d2ef59bc4df2c503b7d92', 'dba1804b1d180eda2bf0439952e9706782109fd6'

def git(*a, raw=False):
    out = subprocess.run(['git', '-C', REPO, *a], capture_output=True, check=True).stdout
    return out if raw else out.decode('utf-8', 'surrogateescape')

def tree(c):
    d = {}
    for line in git('ls-tree', '-r', '-z', '--full-tree', c).split('\0'):
        if not line:
            continue
        meta, path = line.split('\t', 1)
        mode, kind, oid = meta.split()
        d[path] = (mode, kind, oid)
    return d

def blob(c, p):
    return git('show', f'{c}:{p}', raw=True)

def sha(b):
    return hashlib.sha256(b).hexdigest()

r = {}
tb, tl, tr, tm = tree(BASE), tree(LEFT), tree(RIGHT), tree(MERGE)
changed_l = {p for p in set(tb) | set(tl) if tb.get(p) != tl.get(p)}
changed_r = {p for p in set(tb) | set(tr) if tb.get(p) != tr.get(p)}
r['leftChanged'] = len(changed_l)
r['rightChanged'] = len(changed_r)
r['overlap'] = sorted(changed_l & changed_r)
bad = []
counts = {'left': 0, 'right': 0, 'unchangedBase': 0}
allpaths = set(tb) | set(tl) | set(tr) | set(tm)
for p in sorted(allpaths):
    if p in changed_l:
        exp, side = tl.get(p), 'left'
    elif p in changed_r:
        exp, side = tr.get(p), 'right'
    else:
        exp, side = tb.get(p), 'unchangedBase'
    if tm.get(p) != exp:
        bad.append((p, side, exp, tm.get(p)))
    elif exp is not None:
        counts[side] += 1
r['mergeTreeEntries'] = len(tm)

r['mergeMismatches'] = bad
# deletions?
