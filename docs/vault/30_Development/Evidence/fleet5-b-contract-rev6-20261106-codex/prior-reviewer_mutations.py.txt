"""Claude reviewer-owned in-memory Git source mutations against owner Rev5 analyzers.

Read-only: imports owner analyzers from the immutable 228e evidence directory, mutates
an in-memory copy of 72028 Git sources, and prints semantic projections. Writes only
to the path given as argv[1]. No application import, DB, Node or network.
"""

base_map = rc.mapping(baseline, matrix)

def mutate(path, old, new):
    out = dict(sources)
    raw = out[path].decode('utf-8-sig')
    assert raw.count(old) >= 1, (path, old)
    out[path] = raw.replace(old, new, 1).encode()
    return out


def add(path, body):
    out = dict(sources)
    out[path] = body.encode()
    return out


def project(got, path_filter):
    m = rc.mapping(got, matrix)
    errs = rc.audit(got, matrix, m)
    writers = [dict(path=w['path'], line=w['line'], function=w['function'], table=w['table'], verb=w['verb'],
                    columns=w['columns'], disposition=w['disposition'], replacement=w['replacementProcedures'],
                    expression=w['expression'][:160]) for w in m if path_filter(w)]
    recs = [dict(function=r['function'], line=r['line'], tables=r['tables'], modes=r['modes'],
                 scope=r.get('scopeDisposition'), unresolved=r.get('unresolvedSQL'))
            for r in got['records'] if path_filter(r)]
    edges = [dict(function=e['function'], spelling=e['spelling'], targets=e['targets']) for e in got['edges'] if path_filter(e)]
    unknown = [dict(function=e['function'], spelling=e['spelling']) for e in got['unknownEdges'] if path_filter(e)]
    plans = [dict(function=p['function'], locks=p['rankedResourceLocks'], unresolved=len(p['unresolvedDbSites']))
             for p in got['functionPlans'] if path_filter(p)]
    return dict(auditErrors=errs, writers=writers, records=recs, edges=edges, unknownEdges=unknown, plans=plans,
                totalMatrixWriters=len(m))
