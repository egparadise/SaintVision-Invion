def sanity(path, src, skip=()):
    tree = ast.parse(src); k = 0
    for w in WIT:
        if w['path'] != path or tuple(w['span']) in skip: continue
        l, c, el, ec = w['span']; node = next((n for n in ast.walk(tree) if getattr(n, 'lineno', None) == l and n.col_offset == c and n.end_lineno == el and n.end_col_offset == ec), None)
        if node is None or sha(ast.get_source_segment(src, node)) != w['expressionSHA256']: return False, k
        k += 1
    return True, k
def unbound(src):
    """Free Load names at module/function level that are neither builtins nor bound anywhere in the file (coarse, conservative)."""
    import builtins
    tree = ast.parse(src); bound = set(dir(builtins))
    for n in ast.walk(tree):
        if isinstance(n, (ast.Import, ast.ImportFrom)): bound |= {(a.asname or a.name).split('.')[0] for a in n.names}
        elif isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)): bound.add(n.name)
        elif isinstance(n, ast.arg): bound.add(n.arg)
        elif isinstance(n, ast.Name) and isinstance(n.ctx, (ast.Store, ast.Del)): bound.add(n.id)
        elif isinstance(n, ast.ExceptHandler) and n.name: bound.add(n.name)
    return sorted({n.id for n in ast.walk(tree) if isinstance(n, ast.Name) and isinstance(n.ctx, ast.Load) and n.id not in bound})
