import ast,subprocess,sys
CP='services/control-plane/src/inv/'
paths=[CP+n for n in ['observation.py','db.py','errors.py','observer_worker.py','workspace_api.py','workspace_config.py','app.py','worker.py']]+['tools/lan_pilot.py','tools/remote_workspace_entry.py','tools/lan_execution.py']
out={}
for p in paths:
    s=subprocess.check_output(['git','show','72028cbac903c4415d9979f980a6e0f9948e914f:'+p]).decode('utf-8-sig');t=ast.parse(s)
    defs={n.name for n in ast.walk(t) if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef))}
    news=[n.lineno for n in ast.walk(t) if isinstance(n,ast.FunctionDef) and n.name in ('__new__','__call__')]
    meta=[n.lineno for n in ast.walk(t) if isinstance(n,ast.ClassDef) and n.keywords]
    rebind=[(n.lineno,x.id) for n in t.body if isinstance(n,(ast.Assign,ast.AugAssign,ast.AnnAssign)) for x in (n.targets if isinstance(n,ast.Assign) else [n.target]) if isinstance(x,ast.Name) and x.id in defs]
    glob=[n.lineno for n in ast.walk(t) if isinstance(n,ast.Global)]
    deco=sorted({ast.unparse(d) for n in ast.walk(t) if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef)) for d in n.decorator_list})
    out[p]={'__new__/__call__':news,'classKeywords':meta,'moduleRebindOfDefName':rebind,'global':glob,'decorators':deco}
for k,v in out.items():print(k,v)
