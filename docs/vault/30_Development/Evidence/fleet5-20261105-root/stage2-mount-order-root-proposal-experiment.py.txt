def mounts_vector(mounts):
    """Only the supported plain, nonoverlapping mount destinations; retain ALL dictionary fields."""
    if type(mounts) is not list or not 1 <= len(mounts) <= 2:
        raise ValueError('ambiguous plain mount topology')
    seen=set()
    for m in mounts:
        if type(m) is not dict or any(type(k) is not str for k in m):
            raise ValueError('typed mount required')
        destination=m.get('Destination')
        if type(destination) is not str or destination not in {'/state','/var/run/docker.sock'} or destination in seen:
            raise ValueError('unknown/duplicate mount destination')
        seen.add(destination)
        if m.get('Type') != ('volume' if destination=='/state' else 'bind'):
            raise ValueError('plain mount type differs')
        for key in ('Name','Source','Driver','Mode','Propagation'):
            if key in m and type(m[key]) is not str:
                raise ValueError('non-string mount field')
        if type(m.get('RW')) is not bool or not m['RW']:
            raise ValueError('plain writable state/socket type differs')
    if '/state' not in seen:
        raise ValueError('state mount missing')
    # Exact existing dictionaries, not hashes or projected field allowlists.
    return tuple(sorted(mounts,key=lambda m:m['Destination']))
