import os,stat,sys
from pathlib import Path
try:
    p=Path('/run/secrets/runtime-dsn')
    s=p.lstat()
    if not stat.S_ISREG(s.st_mode) or s.st_uid!=os.geteuid() or s.st_mode&0o077 or s.st_size>16384:
        raise ValueError()
    value=p.read_text(encoding='utf-8').strip()
    if not value:
        raise ValueError()
    for key in list(os.environ):
        if key.startswith('PG'):
            os.environ.pop(key,None)
    os.environ['INV_RUNTIME_DSN']=value
    del value
    from inv.observer_worker import main
    main()
except BaseException:
    raise SystemExit('Explicit scoped observer unavailable') from None
