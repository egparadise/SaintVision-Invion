def guard_location(path, source, error):  # diagnostic only: first refused node, same args as census
    f = Facts(path, source); allowed = u.baseline_sites(path, True, (), ()).copy()
    for n, tag in u.audited_nodes(f, True, (), {}, (), ()):
        if u.class_field_factory_allowed(f, n): continue
        k = u.key_for(f, n, tag)
        if allowed[k] <= 0: return {'tag': tag, 'span': [n.lineno, n.col_offset, n.end_lineno, n.end_col_offset]}
        allowed[k] -= 1
    return None
