def git(*args):
    return subprocess.run(["git", *args], capture_output=True, check=True).stdout


def blob(path):
    return git("show", f"{CAND}:{path}")


def imports(tree):
    found = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            found += [a.name for a in node.names]
        elif isinstance(node, ast.ImportFrom):
            found.append(("." * node.level) + (node.module or ""))
    return sorted(set(found))


def call_names(tree):
    names = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Call):
            f = node.func
            parts = []
            while isinstance(f, ast.Attribute):
                parts.append(f.attr)
                f = f.value
            if isinstance(f, ast.Name):
                parts.append(f.id)
            names.add(".".join(reversed(parts)))
    return sorted(names)


def sql_literals(tree):
    rows = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Constant) and isinstance(node.value, str):
            text = " ".join(node.value.split())
            upper = text.upper()
            if any(k in upper for k in ("SELECT ", "INSERT INTO", "UPDATE ", "DELETE ")) and (
                    "inv." in text or "public." in text or "pg_catalog" in text):
                rows.append({"line": node.lineno, "sql": text,
                             "tenantPredicateOrValue": "tenant_id" in text})
    return rows


def decorator_name(dec):
    target = dec.func if isinstance(dec, ast.Call) else dec
    parts = []
    while isinstance(target, ast.Attribute):
        parts.append(target.attr)
        target = target.value
    if isinstance(target, ast.Name):
        parts.append(target.id)
    return ".".join(reversed(parts))


def test_cases(tree):
    functions = []
    for node in tree.body:
        if isinstance(node, ast.FunctionDef) and node.name.startswith("test_"):
            cases, postgres = 1, False
            for dec in node.decorator_list:
                name = decorator_name(dec)
                if name.endswith("mark.postgres"):
                    postgres = True
                if name.endswith("mark.parametrize") and isinstance(dec, ast.Call):
                    values = dec.args[1]
                    if isinstance(values, (ast.List, ast.Tuple)):
                        cases *= len(values.elts)
                    else:
                        cases = None
            args = [a.arg for a in node.args.args]
            functions.append({"name": node.name, "line": node.lineno, "cases": cases,
                              "postgresMark": postgres, "usesFreshDb": "fresh_db" in args})
    return functions


def main():
    names = git("diff", "--name-status", "--no-renames", BASE, CAND).decode().splitlines()
    statuses = [line.split("\t", 1) for line in names]
    parent = git("rev-parse", f"{CAND}^").decode().strip()
    result = {
        "candidate": CAND, "directParent": parent, "directParentMatches": parent == BASE,
        "deltaPathCount": len(statuses),
        "deltaStatusSet": sorted({s for s, _ in statuses}),
        "nonDocPaths": sorted(p for s, p in statuses
                              if not (p.startswith(EVIDENCE_PREFIX) or p == HISTORY)),
        "ownDocPathCount": sum(1 for s, p in statuses
                               if p.startswith(EVIDENCE_PREFIX) or p == HISTORY),
        "sources": {},
    }
    result["onlyFourSourcesPlusOwnDocsAllAdded"] = (
        result["deltaStatusSet"] == ["A"] and result["nonDocPaths"] == sorted(SOURCES))
    totals = {"functions": 0, "cases": 0, "postgresFreshDbCases": 0}
    for path, expected in SOURCES.items():
        raw = blob(path)
        tree = ast.parse(raw.decode("utf-8"), filename=path)
        entry = {"bytes": len(raw), "sha256": hashlib.sha256(raw).hexdigest(),
                 "expectedSha256Matches": hashlib.sha256(raw).hexdigest() == expected,
                 "imports": imports(tree), "calls": call_names(tree)}
        if path.startswith("tests/"):
            cases = test_cases(tree)
            entry["testFunctions"] = cases
            totals["functions"] += len(cases)
            totals["cases"] += sum(c["cases"] for c in cases)
            totals["postgresFreshDbCases"] += sum(
                c["cases"] for c in cases if c["postgresMark"] and c["usesFreshDb"])
        else:
            entry["sqlLiterals"] = sql_literals(tree)
        result["sources"][path] = entry
    result["testTotalsFromAST"] = totals
    raw_out = (json.dumps(result, indent=1, sort_keys=True, ensure_ascii=False) + "\n").encode()
    OUT.write_bytes(raw_out)
    print(json.dumps({"out": OUT.name, "sha256": hashlib.sha256(raw_out).hexdigest(),
                      "totals": totals, "scopeOK": result["onlyFourSourcesPlusOwnDocsAllAdded"],
                      "hashesOK": all(v["expectedSha256Matches"] for v in result["sources"].values())}))


if __name__ == "__main__":
    main()
