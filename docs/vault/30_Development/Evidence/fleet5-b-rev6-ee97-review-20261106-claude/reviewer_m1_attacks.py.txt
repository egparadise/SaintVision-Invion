import model as m
from authority import OP,VER,RUN,SAFE,REV

def rec(name,actual,expected_by_contract,note):
    R.append(dict(case=name,actual=actual,contractExpectation=expected_by_contract,defect=actual!=expected_by_contract,note=note))

s=m.scope(); s2=m.scope(n='node-2')
# A1 read-only heartbeat transient failure
w=m.World();r=w.connect(RUN);v=w.connect(VER);op=w.connect(OP)
b=m.binding(w,op='hb-1',kind='HEARTBEAT_OBSERVE');h=w.register(r,b)

rec('A1-inflight-heartbeat-denies-business',w.eligible('R03',s),True,'no side effect yet business denied for the whole Node during probe')
w.invoke(h,b['bodySHA'],b['method'],b['rawPath']);w.failure(h)  # 3s timeout, routine network loss
st=w.obligations[(m.sk(s),'hb-1')]['state']
rec('A1-heartbeat-timeout-state',st,'NO_EFFECT_ABANDONED_or_ordinary','read-only IO uncertainty has no physical effect')
rec('A1-after-timeout-all13-readers',[w.eligible(x,s) for x in w.readers].count(True),13,'Node business stays denied until OP')
w2=deepcopy(w)
rec('A1-ordinary-verifier-cannot-recover',w.confirm(v,h,w.outcome(h,'OBSERVED')),True,'ordinary path unavailable after read-only failure')
b2=m.binding(w,op='hb-2',kind='HEARTBEAT_OBSERVE');h2=w.register(r,b2);w.invoke(h2,b2['bodySHA'],b2['method'],b2['rawPath'])
rec('A1-next-good-heartbeat-confirms',w.confirm(v,h2,w.outcome(h2,'OBSERVED')),True,'')
rec('A1-good-heartbeat-does-not-restore',w.eligible('R03',s),True,'only OP physical proof clears read-only timeout')
rec('A1-needs-OP-physical-proof',w.settle(op,h,phys(h)),False,'operator escalation for a no-effect read')
# A2 renewal signing denied while any read-only observation in flight / failed
w=m.World();op=w.connect(OP);r=w.connect(RUN)
b=m.binding(w,op='hb-x',kind='HEARTBEAT_OBSERVE');h=w.register(r,b);w.invoke(h,b['bodySHA'],b['method'],b['rawPath']);w.failure(h)
rec('A2-same-config-renew-after-heartbeat-timeout',w.sign(op,s,'APPROVAL',m.sign_body(w,kind='APPROVAL')) is not None,True,'renewal blocked until OP physical recovery; may lead to expiry')
# A3 single-flight per Node
w=m.World();r=w.connect(RUN)
h1=w.register(r,m.binding(w,op='exec-1'))
h2=w.register(r,m.binding(w,op='exec-2'))

w=m.World();r=w.connect(RUN);op=w.connect(OP)
w.set_fact(op,s,'DRAIN')
ht=w.register(r,m.binding(w,op='term-1',kind='TERMINAL_FRAME'))
rec('A4-terminal-frame-registration-under-DRAIN',ht is not None,True,'catalog R12 says existing claimed terminal may continue under drain; register() requires eligible(R03) = no facts')
# A5 cross-Node blast radius of one fuse-persistence failure
w=m.World();r=w.connect(RUN);v=w.connect(VER);op=w.connect(OP)
ha=w.register(r,m.binding(w,s=s,op='a',kind='GPU_ACTIVE_OBSERVE'))

rec('A5-node2-eligible',w.eligible('R03',s2),True,'other-scope obligation failure denies node-2')
# A6 exception before invalidate: residual capability exposure check
w=m.World();r=w.connect(RUN);v=w.connect(VER)
