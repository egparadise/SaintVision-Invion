rows = []


def kst():
    return datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=9))).isoformat(timespec='seconds')


def sha(b):
    return hashlib.sha256(b).hexdigest()


def git(*argv):
    return subprocess.run(['git', *argv], cwd=REPO, check=True, capture_output=True).stdout


def row(probe, ok, **facts):
    rows.append({'probe': probe, 'pass': bool(ok), **facts})


def tree_blobs(rev):
    out = git('ls-tree', '-r', '-z', rev).split(b'\0')
    res = {}
    for item in out:
        if not item:
            continue
        meta, path = item.split(b'\t', 1)
        _mode, kind, oid = meta.split()
        if kind == b'blob':
            res[path.decode('utf8')] = oid.decode()
    return res


def cat_batch(oids):
    proc = subprocess.run(['git', 'cat-file', '--batch'], cwd=REPO, input=''.join(o + '\n' for o in oids).encode(),
                          capture_output=True, check=True)
    data, pos, res = proc.stdout, 0, {}
    for oid in oids:
        nl = data.index(b'\n', pos)
        header = data[pos:nl].split()
        size = int(header[2])
        res[oid] = data[nl + 1:nl + 1 + size]
        pos = nl + 1 + size + 1
    return res


def dump(n):
    return ast.dump(n, include_attributes=False)


def once_of(tree):
    cls = [n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == 'DeliveryWorker']
    assert len(cls) == 1
    fn = [n for n in cls[0].body if isinstance(n, ast.FunctionDef) and n.name == 'once']
    assert len(fn) == 1
    return cls[0], fn[0]
