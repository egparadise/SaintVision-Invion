"""Claude: binder-only local-shadow attribution at one producer directory (pure AST, no execution).

Usage: python -B -I reviewer_rev9g_local_shadow_check.py <producer evidence dir (absolute)>
Prints, for the binder fixture b-imported-function-scope-call-shadowed-local and two neighbours, the resolved call
target, caller_error (if the producer has it), unresolved reasons and owned formals.
"""
import ast, json, sys
from pathlib import Path
E = Path(sys.argv[1]).resolve(strict=True); sys.path.insert(0, str(E))
import parameter_flow as pf, call_binding as cb, source_bindings as sb, callee_identity as ci
from forwarding_discovery import canonical, aliases
P = 'fixtures/claude_b9f_binder.py'; PA = 'fixtures/claude_b9f_callee.py'
PRE = 'from inv.workspace_api import WorkspaceAPI as Owner\nfrom foreign_module import receive\nheld = Owner(None, None, None)\n'
DEF = 'def accept(slot):\n    return None\n'
CASES = {
    'b-imported-function-scope-call-shadowed-local': PRE + 'from fixtures.claude_b9f_callee import accept\ndef use():\n    accept = receive\n    return accept(held)\n',
    'b-imported-function-scope-call-shadowed-local-param': PRE + 'from fixtures.claude_b9f_callee import accept\ndef use(accept=receive):\n    return accept(held)\n',
    'b-control-imported-function-scope-call-no-shadow': PRE + 'from fixtures.claude_b9f_callee import accept\ndef use():\n    return accept(held)\n',
}
out = {'producerDir': E.name, 'hasCallerError': hasattr(ci, 'caller_error'), 'cases': []}
for name, body in CASES.items():
    sources = {PA: DEF, P: body}
    for p, s in sources.items(): compile(s, p, 'exec')
    data, error = pf.solve(sources)
    facts = {p: sb.Facts(p, s) for p, s in sorted(sources.items())}
    for f in facts.values(): aliases(f, {})
    functions, classes = cb.registry(facts); f = facts[P]
    call = next(n for n in ast.walk(f.tree) if isinstance(n, ast.Call) and isinstance(n.func, ast.Name) and n.func.id == 'accept')
    symbol, pair, _ = cb.target(f, call, functions, classes, lambda s: canonical(s, {}))
    ce = ci.caller_error(f, call) if hasattr(ci, 'caller_error') else 'N/A (no caller_error in this producer)'
    out['cases'].append({'case': name, 'sourceSHA256': {p: sb.sha(s) for p, s in sorted(sources.items())}, 'resolvedTarget': symbol, 'pairFound': pair is not None,
                         'callerError': ce, 'solveError': error, 'unresolved': [x['reason'] for x in (data or {}).get('unresolved', [])],
                         'ownedFormalsAtAccept': sorted((data or {}).get('parameters', {}).get(PA, {}).get('accept', []))})
print(json.dumps(out, indent=1))
