def supplied_9cea(call, index, name, required=False):
    values = ([call.args[index]] if len(call.args) > index else []) + [k.value for k in call.keywords if k.arg == name]
    if any(k.arg is None for k in call.keywords): values.append(None)
    return [None] if len(values) > 1 else values or ([None] if required else [])

def ablate(which):

def unbound(source):
    tr = ast.parse(source); bound = set(dir(builtins))
    for n in ast.walk(tr):
        if isinstance(n, (ast.Import, ast.ImportFrom)): bound.update((a.asname or a.name).split('.')[0] for a in n.names)
        elif isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)): bound.add(n.name)
        elif isinstance(n, ast.arg): bound.add(n.arg)
        elif isinstance(n, ast.Name) and isinstance(n.ctx, (ast.Store, ast.Del)): bound.add(n.id)
        elif isinstance(n, ast.ExceptHandler) and n.name: bound.add(n.name)
        elif isinstance(n, (ast.MatchAs, ast.MatchStar)) and n.name: bound.add(n.name)
        elif isinstance(n, ast.MatchMapping) and n.rest: bound.add(n.rest)
    return {n.id for n in ast.walk(tr) if isinstance(n, ast.Name) and isinstance(n.ctx, ast.Load) and n.id not in bound}

def witnesses(sources):
    checked = 0
    for w in W:
        src = sources.get(w['path']) or sb.frozen_source(w['path']); tree = ast.parse(src)
        node = next((n for n in ast.walk(tree) if hasattr(n, 'lineno') and [n.lineno, n.col_offset, n.end_lineno, n.end_col_offset] == w['span']), None)
        assert node is not None and sb.sha(ast.get_source_segment(src, node)) == w['expressionSHA256'], ('WITNESS', w)
        checked += 1
    return checked

def must_replace(src, old, new):

def roots_preserved(universe):
    census = cg.current_census(universe); facts = {p: sb.Facts(p, s) for p, s in universe.items() if p in S}
    keys = Counter(cg.root_key(r, facts[r['source']['path']]) for r in census['production'] if r['source']['path'] in facts)

        return err
    ablate(None); err = once()

    abl = {}

    row = {'case': name, 'observed': observed, 'reason': err, 'expected': expect,
           'matchesExpectation': expect is None or observed == expect.split(':', 1)[0] and (':' not in expect or (err or '').startswith(expect.split(':', 1)[1])),

    R.append(row); print(json.dumps({'case': name, 'observed': observed, 'reason': err, 'expected': expect, 'ablations': abl}), flush=True)
