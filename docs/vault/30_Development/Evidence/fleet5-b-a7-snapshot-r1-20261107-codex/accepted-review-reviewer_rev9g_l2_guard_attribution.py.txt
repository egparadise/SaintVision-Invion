"""Claude Rev9g L2 guard-level attribution (pure AST) + two full-provenance follow-ups.

Usage: python -B -I reviewer_rev9g_l2_guard_attribution.py <rev9g evidence dir (absolute)> <out.json>
Part A: for every workspace_config.py overlay used by reviewer_rev9g_delta_probes (same text), call the FINAL
callee_identity.caller_error directly at the three frozen relevant Name call sites (:44 WorkspaceAPI,
:47 RestrictedWorkspaceRuntime, :64 SnapshotStore) and report the reason and binding scope. This separates the
Rev9g caller-binding guard from the older use-site guards that also refuse some overlays.
Part B: two full-provenance runs with the caller-binding guard on and then reverted (callee_identity.caller_error
-> None): an equal-count replaced function-scope SnapshotStore import, and the global-declaration-alone control.
No mutant source is imported or executed (ast.parse/compile only).

core = t.read_batch([CP + n for n in t.CORE] + ['tools/remote_workspace_entry.py']); ctors = t.read_batch(t.MAIN_CTOR_PATHS); S = {**core, **ctors}

def must_replace(src, old, new):
    assert src.count(old) == 1, ('ANCHOR', old[:80]); return src.replace(old, new, 1)

OVERLAYS = {

    'L2-installer-RestrictedWorkspaceRuntime': WCF + INST.replace('WorkspaceAPI', 'RestrictedWorkspaceRuntime'),
    'L2-plain-module-reassignment': WCF + '\nfrom foreign_module import wrap as _claude_wrap\nWorkspaceAPI = _claude_wrap(WorkspaceAPI)\n',
    'L2-same-line-second-import': must_replace(WCF, L9, L9[:-1] + '; from foreign_module import WorkspaceAPI\n'),
    'L2-equal-count-replaced-import (invalid full input: roots)': must_replace(WCF, L9, 'from foreign_module import RestrictedWorkspaceRuntime, WorkspaceAPI\n'),
    'L2-module-walrus-in-comprehension': WCF + '\nfrom foreign_module import wrap as _claude_wrap\n_claude_w = [(WorkspaceAPI := _claude_wrap(x)) for x in (WorkspaceAPI,)]\n',
    'L2-global-del-in-function': WCF + '\n\ndef _claude_drop():\n    global WorkspaceAPI\n    del WorkspaceAPI\n',
    'L2-class-body-global-import': WCF + '\n\nclass _ClaudeBox:\n    global WorkspaceAPI\n    from foreign_module import WorkspaceAPI\n',
    'L2-function-scope-import-blank-line': must_replace(WCF, L62, '        from .workspace_recovery import WorkspaceRecovery\n        from foreign_module import SnapshotStore\n        result.recovery'),
    'L2-function-scope-equal-count-replaced-import': must_replace(WCF, L60, '        from foreign_module import SnapshotStore\n'),
    'L2-module-star-import-workspace_config': WCF + '\nfrom foreign_module import *\n',
    'L2-control-global-declaration-alone': WCF + '\n\ndef _claude_g():\n    global WorkspaceAPI\n    return WorkspaceAPI\n',
    'L2-control-nested-local-same-name': WCF + '\n\ndef _claude_n():\n    WorkspaceAPI = 1\n    return WorkspaceAPI\n',

    'L2-control-module-shadow-of-function-scope-import': WCF + '\nfrom foreign_module import SnapshotStore\n',

L37 = '    profile["executables"] = frozenset(profile["executables"])\n'

SITES = [(44, 'WorkspaceAPI'), (47, 'RestrictedWorkspaceRuntime'), (64, 'SnapshotStore')]
EXTRA = {'L2-local-shadow-in-new-function': [(None, 'WorkspaceAPI')]}
A = []
for name, src in OVERLAYS.items():
    compile(src, WC, 'exec'); f = sb.Facts(WC, src); row = {'overlay': name, 'sourceSHA256': sb.sha(src), 'sites': {}}
    for line, callee in SITES + EXTRA.get(name, []):
        if line is None:
            line = max(n.lineno for n in ast.walk(f.tree) if isinstance(n, ast.Call) and isinstance(n.func, ast.Name) and n.func.id == callee)
        call = next(n for n in ast.walk(f.tree) if isinstance(n, ast.Call) and n.lineno == line and isinstance(n.func, ast.Name) and n.func.id == callee)
        reason = ci.caller_error(f, call); proof = f._caller_binding_proofs[(call.lineno, call.col_offset, call.end_lineno, call.end_col_offset)]
        row['sites'][f'{line}:{callee}'] = {'reason': reason, 'bindingScope': proof['bindingScope'], 'frozenBindingScope': proof['frozenBindingScope'],
                                            'currentEvents': sum(x['count'] for x in proof['currentMultiset']), 'frozenEvents': sum(x['count'] for x in proof['frozenMultiset']),
                                            'stars': sum(x['count'] for x in proof['currentStars'])}

# Part B: full provenance follow-ups, guard on then reverted

FROZEN_ROOTS = cg.frozen_root_keys()
def unbound(source):
    tr = ast.parse(source); bound = set(dir(builtins))
    for n in ast.walk(tr):
        if isinstance(n, (ast.Import, ast.ImportFrom)): bound.update((a.asname or a.name).split('.')[0] for a in n.names)
        elif isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)): bound.add(n.name)
        elif isinstance(n, ast.arg): bound.add(n.arg)
        elif isinstance(n, ast.Name) and isinstance(n.ctx, (ast.Store, ast.Del)): bound.add(n.id)
        elif isinstance(n, ast.ExceptHandler) and n.name: bound.add(n.name)
    return {n.id for n in ast.walk(tr) if isinstance(n, ast.Name) and isinstance(n.ctx, ast.Load) and n.id not in bound}

        assert node is not None and sb.sha(ast.get_source_segment(s, node)) == w['expressionSHA256']; checked += 1
    census = cg.current_census(changed); facts = {p: sb.Facts(p, s) for p, s in changed.items()}

    for mode in ('final', 'caller_error-reverted'):
        ci.caller_error = ORIG if mode == 'final' else (lambda f, call: None)
        _, err = t.provenance({p: changed[p] for p in core}, {p: changed[p] for p in ctors})
        res[mode] = {'observed': 'ACCEPT' if err is None else 'REFUSE', 'reason': err}
    ci.caller_error = ORIG
    Bf.append({'case': name, 'sourceSHA256': sb.sha(src), 'witnessesChecked': checked, 'sixFrozenRootsPreserved': preserved, 'newUnboundNames': [], **res})
    print(json.dumps(Bf[-1]), flush=True)
# a brand-new production path (no frozen baseline -> binder-only caller mode inside full provenance)
NP = CP + 'claude_new_caller.py'

try: sb.frozen_source(NP); new_has_baseline = True
