function honor<T>(signal: AbortSignal | undefined, body: (res: (v: T) => void, rej: (e: any) => void) => void): Promise<T> {
  return new Promise<T>((res, rej) => {
    if (signal) signal.addEventListener('abort', () => rej(signal.reason ?? new DOMException('Aborted', 'AbortError')));
    body(res, rej);
  });
}

describe('UIR2 reviewer probes: Header gateway', () => {
  const renderHeader = async () => { await act(async () => { root.render(<Header currentTheme="light" onToggleTheme={() => {}} activeTab="dashboard" onSelectTab={() => {}} />); }); };
  const label = () => (box.querySelector('[title*="Gateway"]')?.textContent ?? text()).replace(/\s+/g, ' ');

  it('H3 header JSON body hangs after headers: deadline does not cover res.json()', async () => {
    let n = 0;
    vi.spyOn(globalThis, 'fetch').mockImplementation((async (_u: any, init?: RequestInit) => {
      n++;
      if (n === 1) return new Response(JSON.stringify({ status: 'ok' }), { status: 200, headers: { 'content-type': 'application/json' } });
      const sig = init?.signal;
      return { ok: true, status: 200, headers: new Headers({ 'content-type': 'application/json' }),
        json: () => new Promise((_r, rej) => { sig?.addEventListener('abort', () => rej(sig.reason ?? new Error('aborted'))); }) } as any;
    }) as any);
    await renderHeader(); await advance(0);
    const first = label();
    await advance(60000);
    const after60 = label();
    report('H3_hungBody', { first, after60, checks: n });
    expect(/\d+ms/.test(after60)).toBe(false);
  });

  it('H4 header fetch hangs (signal honoured): 4s deadline -> Offline, never indefinite 확인 중', async () => {
    vi.spyOn(globalThis, 'fetch').mockImplementation(((_u: any, init?: RequestInit) => honor(init?.signal ?? undefined, () => {})) as any);
    await renderHeader(); await advance(0);
    const at0 = label();
    await advance(4100);
    const at4 = label();
    report('H4_timeout', { at0, at4 });
    expect(at4).toContain('Offline');
  });

  it('H5 header order: degraded -> healthy -> network, each latest wins', async () => {
    const seq = [
      () => new Response(JSON.stringify({ status: 'not_ready' }), { status: 503, headers: { 'content-type': 'application/json' } }),
      () => new Response(JSON.stringify({ status: 'ok' }), { status: 200, headers: { 'content-type': 'application/json' } }),
      () => { throw new TypeError('Failed to fetch'); },
    ];
    let i = 0;
    vi.spyOn(globalThis, 'fetch').mockImplementation((async () => seq[Math.min(i++, 2)]()) as any);
    await renderHeader(); await advance(0);
    const a = label(); await advance(5000); const b = label(); await advance(5000); const c = label();
    report('H5_order', { a, b, c });
    expect(a).toContain('비정상 응답'); expect(b).toMatch(/\d+ms/); expect(c).toContain('Offline');
  });
});
