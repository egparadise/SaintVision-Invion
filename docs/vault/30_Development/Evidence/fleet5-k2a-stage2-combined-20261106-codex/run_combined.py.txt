"""Authorized serial combined-source checks; no fixture/live provisioning."""
from pathlib import Path
import datetime
import json
import os
import subprocess
import sys

root = Path.cwd()
out = root / 'docs/vault/30_Development/Evidence/fleet5-k2a-stage2-combined-20261106-codex'
env = dict(os.environ, GOTOOLCHAIN='local', GOWORK='off', GOPROXY='off',
           GOSUMDB='off', GOMAXPROCS='1', PYTHONDONTWRITEBYTECODE='1')
checks = [
    ('combined-go', ['go', 'test', '-p', '1', '-count=1', '-v', './internal/migrationcontext'], root / 'services/node-agent'),
    ('combined-vet', ['go', 'vet', '-p', '1', './internal/migrationcontext', './migrationverify'], root / 'services/node-agent'),
    ('standalone-unedited', [sys.executable, '-B', '-m', 'pytest', '-p', 'no:cacheprovider', '-q', '-ra', 'tests/test_lan_leaf_rotation_standalone_runtime.py'], root),
    ('stage2-controlled', [sys.executable, '-B', '-m', 'pytest', '-p', 'no:cacheprovider', '-q', '-ra', 'tests/core/test_worker_peer_floor_observation_diagnostics.py', 'tests/core/test_worker_peer_floor_mount_order_stage2.py', 'tests/core/test_lan_workspace_upgrade.py'], root),
]
report = dict(classification='LOCAL_COMBINED_SOURCE_ONLY_NOT_CORE_RUNTIME_OR_PL_REVIEW_CLOSURE',
              sourceSHA=subprocess.check_output(['git', 'rev-parse', 'HEAD']).decode().strip(),
              platform=sys.platform, python=sys.version,
              goVersion=subprocess.check_output(['go', 'version'], env=env).decode().strip(),
              cgoEnabled=subprocess.check_output(['go', 'env', 'CGO_ENABLED'], env=env).decode().strip(),
              race='NOT_RUN_CGO0', provisionedLANAcceptance='NOT_RUN_NO_LOCAL_DOCKER',
              runtime40Cases='NOT_RUN', runs=[])
for name, argv, cwd in checks:
    start = datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=9))).isoformat()
    print(json.dumps(dict(starting=name, argv=argv, kst=start)), flush=True)
    result = subprocess.run(argv, cwd=cwd, env=env, capture_output=True)
    raw = result.stdout + result.stderr
    (out / (name + '.log')).write_bytes(raw)
    text = raw.decode(errors='replace')
    item = dict(name=name, argv=argv, cwd=str(cwd), exit=result.returncode,
                startedKST=start, endedKST=datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=9))).isoformat(), log=name+'.log')
    if name == 'combined-go':
        item.update(topLevelPass=sum(x.startswith('--- PASS:') for x in text.splitlines()),
                    subtestsPass=sum(x.startswith('    --- PASS:') for x in text.splitlines()),
                    actualMutantBuildExit0Count=text.count('actual go build -p=1 ./internal/migrationcontext exit0;'))
    report['runs'].append(item)
    (out / 'local-combined-results.json').write_text(json.dumps(report, indent=2)+'\n')
    print(json.dumps(item), flush=True)
    if result.returncode:
        print(text[-6000:], flush=True)
        sys.exit(result.returncode)
    if name in ('standalone-unedited', 'stage2-controlled'):
        print(text[-5000:], flush=True)
