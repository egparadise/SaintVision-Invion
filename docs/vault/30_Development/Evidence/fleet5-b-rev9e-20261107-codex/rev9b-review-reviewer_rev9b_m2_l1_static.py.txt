def sanity(path, mutated, intentionally_changed=()):
    """Fresh AST + every recorded witness in this file still resolves (span + expression hash), except spans the case deliberately rewrites."""
    tree = ast.parse(mutated); checked = 0
    for w in WITNESS:
        if w['path'] != path or tuple(w['span']) in intentionally_changed: continue
        l, c, el, ec = w['span']
        node = next((n for n in ast.walk(tree) if getattr(n, 'lineno', None) == l and n.col_offset == c and n.end_lineno == el and n.end_col_offset == ec), None)
        if node is None or sha(ast.get_source_segment(mutated, node)) != w['expressionSHA256']: return False, checked
        checked += 1
    return True, checked
