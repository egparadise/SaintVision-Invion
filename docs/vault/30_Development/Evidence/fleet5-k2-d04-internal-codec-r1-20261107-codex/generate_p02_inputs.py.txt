# Include generated wire types when emitting safe budget/freeze routines.
wire=(D/'p02_wire_internal.go').read_text()
import re
alltypes=dict(types)
for n,body in re.findall(r'type p02(\w+) struct \{([^}]+)\}',wire):
 fields=[]
 for line in body.strip().splitlines():
  k,v=line.split();fields.append((k,v))
 alltypes[n]=fields

def clone_expr(v,t):
 if t in ('string','int64','bool','byte'):return v
 if t.startswith('p02'):return f'p02Freeze{t[3:]}({v})'
 raise AssertionError(t)
def budget_expr(v,t):
 if t=='string':return f'p02BudgetString({v},b)'
 if t in ('int64','bool','byte'):return f'p02Budget{t.title()}({v},b)'
 if t.startswith('p02'):return f'p02Budget{t[3:]}({v},b)'
 raise AssertionError(t)
for n,fields in alltypes.items():
 f=[f'func p02Freeze{n}(v p02{n}) p02{n} {{out:=v'];budget=[f'func p02Budget{n}(v p02{n},b *p02Budget) bool {{']
 for k,t in fields:
  v='v.'+k;o='out.'+k
  if t=='[]byte':f+=[f'{o}=copyBytes({v})'];budget+=[f'if len({v})>p02PayloadLimit||!b.take(int64(len({v}))) {{return false}}']
  elif t.startswith('[]'):
   item=t[2:];f+=[f'{o}=make({t},len({v}));for i,x:=range {v} {{{o}[i]={clone_expr("x",item)}}}'];budget+=[f'if len({v})>4096 {{return false}};for _,x:=range {v} {{if !{budget_expr("x",item)} {{return false}}}}']
  elif t=='map[string]string':
   f+=[f'{o}=copyMap({v})'];budget+=[f'if len({v})>4096 {{return false}};for k,x:=range {v} {{if !p02BudgetString(k,b)||!p02BudgetString(x,b) {{return false}}}}']
  elif t=='map[string]map[string]string':
   f+=[f'{o}=make({t},len({v}));for k,x:=range {v} {{{o}[k]=copyMap(x)}}'];budget+=[f'if len({v})>4096 {{return false}};for k,x:=range {v} {{if !p02BudgetString(k,b)||len(x)>4096 {{return false}};for a,z:=range x {{if !p02BudgetString(a,b)||!p02BudgetString(z,b) {{return false}}}}}}']
  elif t=='map[string]p02Pin':
   f+=[f'{o}=copyMap({v})'];budget+=[f'if len({v})>4096 {{return false}};for k,x:=range {v} {{if !p02BudgetString(k,b)||!p02BudgetPin(x,b) {{return false}}}}']
  elif t.startswith('*p02'):
   sub=t[4:];f+=[f'if {v}!=nil {{x:=p02Freeze{sub}(*{v});{o}=&x}}'];budget+=[f'if {v}!=nil&&!p02Budget{sub}(*{v},b) {{return false}}']
  else:f+=[f'{o}={clone_expr(v,t)}'];budget+=[f'if !{budget_expr(v,t)} {{return false}}']
 f+=['return out}'];budget+=['return true}'];s+=f+budget
