def unbound(source):
    tr = ast.parse(source); bound = set(dir(builtins))
    for n in ast.walk(tr):
        if isinstance(n, (ast.Import, ast.ImportFrom)): bound.update((a.asname or a.name).split('.')[0] for a in n.names)
        elif isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)): bound.add(n.name)
        elif isinstance(n, ast.arg): bound.add(n.arg)
        elif isinstance(n, ast.Name) and isinstance(n.ctx, (ast.Store, ast.Del)): bound.add(n.id)
        elif isinstance(n, ast.ExceptHandler) and n.name: bound.add(n.name)
    return {n.id for n in ast.walk(tr) if isinstance(n, ast.Name) and isinstance(n.ctx, ast.Load) and n.id not in bound}

def witnesses(sources, skip):
    checked = 0
    for w in W:
        if (w['path'], tuple(w['span'])) in skip: continue
        src = sources.get(w['path']) or sb.frozen_source(w['path']); tree = ast.parse(src)
        node = next((n for n in ast.walk(tree) if hasattr(n, 'lineno') and [n.lineno, n.col_offset, n.end_lineno, n.end_col_offset] == w['span']), None)
        assert node is not None and sb.sha(ast.get_source_segment(src, node)) == w['expressionSHA256'], ('WITNESS', w)
        checked += 1
    return checked

def must_replace(src, old, new):
    assert src.count(old) == 1, ('ANCHOR', old[:80], src.count(old)); return src.replace(old, new, 1)

def run(name, changes=None, world=None, expect=None, skip=(), extra_core=None, note=None, graph_only=False):
    changes = changes or {}; world = world or {}
    changed = {**S, **changes}
    for p, s in {**changes, **world, **(extra_core or {})}.items():
        ast.parse(s)
        try: base = sb.frozen_source(p)
        except Exception: base = ''
        nb = sorted(unbound(s) - unbound(base)); assert not nb, (name, nb)
    checked = witnesses({**world, **changed}, skip)
    c2 = {p: changed[p] for p in core}; c2.update(extra_core or {}); k2 = {p: changed[p] for p in ctors}
    if graph_only:
        out, err = cg.prove({**c2, **k2}, world or None)
    else:
        out, err = t.provenance(c2, k2, universe_overrides=world or None)
    universe = {**world, **changed, **(extra_core or {})}
    census = cg.current_census(universe)
    disc, derr = fd.discover(census['production'], universe)
    observed = 'ACCEPT' if err is None else 'REFUSE'
    row = {'case': name, 'observed': observed, 'reason': err, 'expected': expect, 'matchesExpectation': expect is None or observed == expect.split(':', 1)[0] and (':' not in expect or (err or '').startswith(expect.split(':', 1)[1])),
           'mode': 'cg.prove graph only' if graph_only else 't1_provenance.provenance (graph + use sites)',
           'astValid': True, 'newUnboundNames': [], 'witnessesChecked': checked, 'intentionalRootRewrites': sorted(map(list, skip)),
           'mutatedSHA256': {p: sb.sha(s) for p, s in sorted({**changes, **world, **(extra_core or {})}.items())},
           'visibleProductionRoots': len(census['production']), 'discoveryError': derr,
           'discoveredProductionFiles': disc['productionFiles'] if disc else None, 'discoveredHarnessFiles': disc['harnessFiles'] if disc else None,
           'trackedForwarders': disc['trackedForwarders'] if disc else None, 'note': note}
    R.append(row); print(json.dumps({'case': name, 'observed': observed, 'reason': err, 'expected': expect}), flush=True)
    return row
