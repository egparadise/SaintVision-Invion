"""Reviewer serial local publication gates + canonical sync CHECK (argv only, shell=False, no apply/state write).

Usage: python -B reviewer_gate_driver.py.txt <base-ref> <out.txt> <scratch-dir>
Records, for EVERY sync CHECK call, the FULL canonical state file SHA256 immediately before and after.
Same shape as the H12 R2 review driver (381fbbc8), with an H12 6d7 code-review log prefix and a staged
whitespace diff check. Stops at the first nonzero exit (immediate exit), so later gates are not run after a failure.
"""
import datetime, hashlib, subprocess, sys
from pathlib import Path
ROOT = Path(r'D:\Project\SaintVisionI-Invion\fleet5-k2-h12-6d7-code-review-20261107')
ST = Path('D:/Project/SaintVision-Invion/SaintVisionI-Invion/.git/obsidian-sync-state.json')
BASE = sys.argv[1]; OUT = Path(sys.argv[2]); SCR = Path(sys.argv[3])
def kst(): return datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=9))).isoformat(timespec='milliseconds')
def sha(b): return hashlib.sha256(b).hexdigest()
py = lambda *a: [sys.executable, '-B', *a]
sync = py('tools/sync_obsidian.py', '--check', '--state', ST.as_posix(), '--conflicts-out', str(SCR / 'h12c-sync-conflicts.json'))
gates = [py('tools/check_docs.py'), py('tools/check_ontology_generation.py'), py('tools/check_ontology.py'), py('tools/check_contract_bindings.py'),
         py('tools/check_frontend_integrity.py'), py('tools/check_doc_path_citations.py', '--ratchet', '--base-ref', BASE),
         py('tools/check_doc_single_source.py', '--ratchet'), py('tools/check_response_freshness.py'), py('tools/test_sync.py'),
         py('tools/build_docs.py'), sync, ['git', 'diff', '--cached', '--check', BASE]]
lines = ['Reviewer local gates, Windows, D:/Project/SaintVisionI-Invion/.venv Python 3.14.7 -B, PYTHONUTF8=1, cwd=worktree root, serial; tree = ' + BASE[:8] + ' + own staged review History/Evidence only.',
         'sync_obsidian is CHECK only with the canonical Root state; --conflicts-out goes to reviewer scratch. No apply/adopt/resolve/state write/export. Raw logs: scratch h12c-gate-NN.log, sha256 below.']
worst = 0
for i, g in enumerate(gates, 1):
    before = sha(ST.read_bytes()) if g is sync else None
    a = kst(); r = subprocess.run(g, cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.STDOUT); b = kst(); worst = max(worst, r.returncode)
    after = sha(ST.read_bytes()) if g is sync else None
    (SCR / f'h12c-gate-{i:02d}.log').write_bytes(r.stdout)
    shown = ' '.join(['python -B'] * (g[0] == sys.executable) + ['<scratch>/' + Path(x).name if x.startswith(str(SCR)) else x for x in (g[2:] if g[0] == sys.executable else g)])
    lines.append(f'{i:02d} {a}..{b} exit={r.returncode} logsha={sha(r.stdout)} :: {shown}')
    if g is sync: lines.append(f'    stateBefore {before} stateAfter {after}')
    lines += ['    ' + t[:300] for t in r.stdout.decode('utf-8', 'replace').strip().splitlines()[-2:]]
    if r.returncode != 0:
        lines.append(f'STOPPED after gate {i:02d}; later gates not run'); break
lines.append(f'worstExit={worst}')
OUT.write_text('\n'.join(lines) + '\n', encoding='utf-8', newline='\n'); print('\n'.join(lines)); raise SystemExit(worst)
