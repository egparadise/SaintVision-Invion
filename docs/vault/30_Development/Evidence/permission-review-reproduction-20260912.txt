import ast,hashlib,json,subprocess
from pathlib import Path
from types import SimpleNamespace
from uuid import uuid4
import psycopg
pytest_plugins = ["conftest", "test_operational_readiness"]

def test_original_cross_project_comparison(migrated,database_url):
    source=subprocess.check_output(["git","show","d63717f:tools/operational_readiness.py"],text=True,encoding="utf-8")
    tree=ast.parse(source)
    node=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=="snapshot_permissions")
    namespace={"Any":object}
    exec(compile(ast.Module(body=[node],type_ignores=[]),"claude-d63717f", "exec"),namespace)
    tenant=str(uuid4())
    with psycopg.connect(database_url.replace("postgresql+psycopg://","postgresql://")) as c:
        c.execute("INSERT INTO public.tenants(tenant_id,slug,display_name) VALUES(%s,%s,'review fixture')",(tenant,"review-"+tenant))
    args=SimpleNamespace(tenant=tenant,dsn=database_url)
    rows=[]
    for project in ["project-A","project-B","project-A"]:
        row=namespace["snapshot_permissions"](args,dict(projectId=project,userId="synthetic-user",layers=[dict(layer="project",holds=True,value="same-grant")]))
        rows.append(dict(project=project,changed=row["changed"],previousDigest=row["previousDigest"],digest=row["digest"]))
    assert [r["changed"] for r in rows] == [None,True,True]
    assert rows[0]["digest"] == rows[2]["digest"]
    result=dict(sourceSHA="d63717f1d63f00dba3117c0468647ab8a1a2b65f",sourceHash=hashlib.sha256(source.encode()).hexdigest(),scope="original snapshot function against disposable PostgreSQL; reproduces finding, not acceptance",observations=rows,expectedSameProjectComparison=[None,None,False])
    Path(".work/permission-original-reproduction.json").write_text(json.dumps(result,indent=2),encoding="utf-8")


def test_original_disabled_operator_overstates_approval(prepared):
    source=subprocess.check_output(["git","show","d63717f:tools/operational_readiness.py"],text=True,encoding="utf-8")
    namespace={"__name__":"original_review", "__file__":str(Path("tools/operational_readiness.py").resolve())}
    exec(compile(source,"claude-d63717f","exec"),namespace)
    with psycopg.connect(prepared["dsn"]) as c:
        c.execute("UPDATE inv.operator_grants SET enabled=false WHERE tenant_id=%s AND subject_id=%s",(prepared["tenant"],prepared["subject"]))
        result=namespace["grants"](c,prepared["tenant"],prepared["project"],prepared["user"],namespace["_capability_columns"](c))
        layer=next(r for r in result["layers"] if r["layer"]=="operator grants")
        assert layer["holds"] is False
        assert result["mayApproveAsOperator"] is True
    Path(".work/permission-disabled-reproduction.json").write_text(json.dumps(dict(sourceSHA="d63717f1d63f00dba3117c0468647ab8a1a2b65f",scope="original grants function, disposable PostgreSQL, disabled operator",operatorLayerHolds=layer["holds"],mayApproveAsOperator=result["mayApproveAsOperator"],expected=False),indent=2),encoding="utf-8")
