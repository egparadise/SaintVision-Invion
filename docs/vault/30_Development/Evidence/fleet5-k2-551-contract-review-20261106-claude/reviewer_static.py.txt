"""Claude reviewer-owned static source checks for Codex K2 proposal 551d.
Stdlib only; reads Git objects (git show) and the owner Evidence folder.
No network, key, DB, Engine, clock, schedule or runtime access. STATIC only:
not crypto, not a live producer, not runtime acceptance.
argv: <out.json> <owner evidence dir>
"""
import hashlib, json, re, subprocess, sys
from pathlib import Path

OUT, E = Path(sys.argv[1]), Path(sys.argv[2])
BASE = 'eb32b211ab3704935b9431747089ab0715582a74'
REVIEWED = '56651cba697f2ab6fcd7166ca4e6ea11eac28113'
PROPOSAL = '551d7312bb9c806fd423dcf6074158a9802cbb14'
DESIGN = '8874b2dfe0dda9a019bd5f334636668290702e13'
MV = 'services/node-agent/migrationverify/'


def git(*a):
    return subprocess.check_output(['git', *a])


def show(rev, p):
    return git('show', f'{rev}:{p}')


def sha(b):
    return hashlib.sha256(b).hexdigest()


def rd(n):
    return json.loads((E / n).read_text(encoding='utf-8'))


res = {'classification': 'REVIEWER_STATIC_SOURCE_CHECKS_NOT_CRYPTO_NOT_RUNTIME', 'checks': {}}
C = res['checks']

# 1. 28 descriptors: index pins vs raw bytes; contract fields vs owner catalog; slot registry joins.
idx = json.loads(show(BASE, MV + 'resources/proposed-contracts/index.json'))
cat = {c['contractId']: c for c in rd('existing-proof-contract-catalog.json')['contracts']}
rows, bad = [], []
for p in idx['contracts']:
    raw = show(BASE, MV + 'resources/proposed-contracts/' + p['file'])
    m = json.loads(raw)
    sraw = show(BASE, MV + 'resources/proposed-contracts/' + m['schemaFile'])
    rows.append({'id': p['contractId'], 'rawOK': sha(raw) == p['rawSHA256'], 'schemaOK': sha(sraw) == p['schemaSHA256'],
                 'lfOnly': b'\r' not in raw and b'\r' not in sraw})
    o = cat.get(p['contractId'], {})
    for k in ('producerRole', 'signerRole', 'freshness', 'codec', 'signatureDomainASCII', 'requiredVerifier'):
        if o.get(k) != m.get(k):
            bad.append([p['contractId'], k, o.get(k), m.get(k)])
slots = json.loads(show(BASE, MV + 'resources/proof-slot-registry.json'))['entries']
slot_bad = [s['slotId'] for s in slots
            if cat.get(s['contractId'], {}).get('signatureDomainASCII') != s['signatureDomainASCII']
            or cat.get(s['contractId'], {}).get('signerRole') != s['signerRole']
            or cat.get(s['contractId'], {}).get('producerRole') != s['producerRole']]
C['descriptors28'] = {
    'count': len(rows), 'catalogCount': len(cat),
    'allRawAndSchemaPinsMatch': all(r['rawOK'] and r['schemaOK'] for r in rows),
    'allLFOnlyNoCR': all(r['lfOnly'] for r in rows), 'catalogFieldMismatches': bad,
    'slots': len(slots), 'slotDomainRoleProducerMismatches': slot_bad,
    'codecs': sorted({c['codec'] for c in cat.values()}),
    'nativeCount': sum(c['codec'] == 'native-m2-v2' for c in cat.values()),
    'distinctDomains': len({c['signatureDomainASCII'] for c in cat.values()})}

# 2. Seven-file helper closure: literal pins in launcher vs raw bytes at eb32.
L = 'deploy/lan/'
launcher_raw = show(BASE, L + 'leaf_rotation_standalone_runtime.py')
launcher = launcher_raw.decode('utf-8')
block = launcher.split('WORKER_MODULES = {', 1)[1].split('}', 1)[0]
mods = dict(re.findall(r"""["'](\w+\.py)["']:\s*["']([0-9a-f]{64})["']""", block))
lock_pin = re.search(r'LOCK_SHA256 = "([0-9a-f]{64})"', launcher).group(1)
lock_raw = show(BASE, L + 'leaf-rotation-runtime-lock.json')
insp_doc = rd('source-inspection.json')
insp = {f['path']: f['rawSHA256'] for f in insp_doc['files']}
C['helperClosure7'] = {
    'moduleLiteralPins': len(mods),
    'moduleRawMatches': {k: sha(show(BASE, L + k)) == v for k, v in mods.items()},
    'lockRawMatchesLiteral': sha(lock_raw) == lock_pin,
    'launcherRawMatchesOwnerInspection': sha(launcher_raw) == insp.get(L + 'leaf_rotation_standalone_runtime.py'),
    'closureFiles': sorted(list(mods) + ['leaf_rotation_standalone_runtime.py', 'leaf-rotation-runtime-lock.json']),
    'ownerList': sorted(insp_doc['helperClosure']['sevenFileClosure'])}

# 3. Owner source-inspection rehash (independent of the owner validator).
C['ownerInspectionRehash'] = {'count': len(insp), 'mismatches': [p for p, h in insp.items() if sha(show(BASE, p)) != h]}

# 4. A01-A40 vs exact design 8874 original cases.
fut = rd('future-runtime-A01-A40.json')
orig_raw = show(fut['originalSource'], fut['originalPath'])
orig = json.loads(orig_raw)['cases']
diffs = [[o['id'], k] for o, n in zip(orig, fut['cases']) for k, v in o.items() if n.get(k) != v]
C['A01_A40'] = {
    'originalSource': fut['originalSource'], 'originalSourceIsDesign8874': fut['originalSource'] == DESIGN,
    'originalPath': fut['originalPath'], 'originalRawSHA256': sha(orig_raw), 'ownerPinnedSHA': fut['originalRawSHA256'],
    'idsExact': [c['id'] for c in orig] == [f'A{i:02d}' for i in range(1, 41)] == [c['id'] for c in fut['cases']],
    'countOrig': len(orig), 'countNew': len(fut['cases']), 'originalKeyValueDiffs': diffs,
    'originalKeys': sorted({k for c in orig for k in c}),
    'allNotRunNullEvidence': all(c['status'] == 'NOT_RUN' and c['executionEvidence'] is None for c in fut['cases']),
    'addedKeys': sorted({k for c in fut['cases'] for k in c} - {k for c in orig for k in c})}

# 5. NewVerifier occurrences across the whole eb32 tree (all .go), and importers of migrationverify.
gg = git('grep', '-n', 'NewVerifier', BASE, '--', '*.go').decode('utf-8').splitlines()
q = subprocess.run(['git', 'grep', '-l', 'node-agent/migrationverify"', BASE, '--', '*.go'], capture_output=True, text=True)
C['NewVerifier'] = {'allGoOccurrences': len(gg),
                    'nonTest': [l for l in gg if not re.search(r'_test\.go:', l)],
                    'goFilesImportingMigrationverify': q.stdout.splitlines(), 'grepExit': q.returncode}

# 6. 56651 reviewed Go -> eb32 delta, and 551d proposal scope.
d1 = git('diff', '--name-status', REVIEWED, BASE).decode().splitlines()
d2 = git('diff', '--name-status', BASE, PROPOSAL).decode().splitlines()
C['delta56651toEb32'] = {'all': d1, 'migrationverifyOnly': [l for l in d1 if MV in l]}
C['proposal551dVsEb32'] = {
    'count': len(d2), 'nonAdd': [l for l in d2 if not l.startswith('A\t')],
    'outsideOwnerHistoryEvidence': [l for l in d2 if 'fleet5-k2-trusted-producer-contract-20261106-codex/' not in l
                                    and '_FLEET5_K2_trusted_producer_contract_Codex.md' not in l]}

# 7. Producer-matrix existingSource symbol citations actually present in the cited eb32 file.
mx = rd('producer-matrix.json')
missing = []
for r in mx['rows']:
    for ref in r['existingSource']:
        text = show(ref['source'], ref['path']).decode('utf-8', 'replace')
        for tok in re.findall(r'[A-Za-z_][A-Za-z0-9_.]*', ref['symbols']):
            if len(tok) > 3 and (tok[0].isupper() or '_' in tok or re.search(r'[a-z][A-Z]', tok)) and tok not in text:
                missing.append({'row': r['id'], 'path': ref['path'], 'token': tok})
C['matrixSymbolCitations'] = {'rows': len(mx['rows']), 'tokensNotFoundInCitedFile': missing}

# 8. Field placement and the P12 classification of MigrationOperationID.
place = {}
for r in mx['rows']:
    for f in r['inputs']:
        place.setdefault(f, []).append(r['id'])
P12 = next(r for r in mx['rows'] if r['id'] == 'P12')
C['fieldPlacement'] = {'multiRow': {k: v for k, v in place.items() if len(v) > 1 and not k.startswith('@')},
                       'MigrationOperationID': place.get('TrustedContext.MigrationOperationID'),
                       'P12producer': P12['proposedProducer'], 'P12validation': P12['validation']}

# 9. Refusal/state tokens used in contract.md vs the closed catalogs.
gates = rd('authority-and-effect-gates.json')
known = set(gates['errorCatalog']) | set(gates['effectStates'])
contract = (E / 'contract.md').read_text(encoding='utf-8')
toks = set(re.findall(r'\b[A-Z][A-Z0-9]+(?:_[A-Z0-9]+)+\b', contract))
ex = rd('failure-examples.json')['examples']
C['contractUpperSnakeTokensOutsideErrorAndStateCatalogs'] = sorted(toks - known)
C['failureExamples'] = {'count': len(ex), 'ids': [x['id'] for x in ex],
                        'refusalsNotInCatalog': [x['id'] for x in ex if x['expectedRefusal'] not in gates['errorCatalog']]}
C['catalogCodesUnusedByAnyRowOrExample'] = sorted(set(gates['errorCatalog']) - {r['onFailure'] for r in mx['rows']}
                                                  - {x['expectedRefusal'] for x in ex})
OUT.write_text(json.dumps(res, indent=1, ensure_ascii=False) + '\n', encoding='utf-8', newline='\n')
print(json.dumps(C, indent=1, ensure_ascii=False))
