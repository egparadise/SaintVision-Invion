def blobs(commit, paths):
    """Read many blobs with one git cat-file --batch process."""
    p = subprocess.Popen(['git', 'cat-file', '--batch'], cwd=ROOT, stdin=subprocess.PIPE, stdout=subprocess.PIPE)
    out = {}
    for path in paths:
        p.stdin.write(f'{commit}:{path}\n'.encode()); p.stdin.flush()
        header = p.stdout.readline().split()
        size = int(header[2]); data = p.stdout.read(size); p.stdout.read(1)
        out[path] = data
    p.stdin.close(); p.wait()
    return out
