def stamp():
    return datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=9))).isoformat()

def run(name, argv, cwd, e, expected, count=None, reason=None):
    start = stamp()
    print(json.dumps(dict(starting=name, argv=argv, kst=start)), flush=True)
    p = subprocess.run(argv, cwd=cwd, env=e, capture_output=True)
    raw = p.stdout + p.stderr
    (out/(name+'.log')).write_bytes(raw)
    item = dict(name=name, argv=argv, cwd=str(cwd), exit=p.returncode, expectedExit=expected,
                startedKST=start, endedKST=stamp(), log=name+'.log',
                actualMutantBuildExit0=raw.count(b'actual go build -p=1 ./internal/migrationcontext exit0;'),
                intendedReason=reason, reasonPresent=None if reason is None else reason.encode() in raw)
    report['runs'].append(item)
    (out/'proof-results.json').write_text(json.dumps(report, indent=2)+'\n')
    print(json.dumps(item), flush=True)
    if p.returncode != expected or count is not None and item['actualMutantBuildExit0'] != count or reason is not None and not item['reasonPresent']:
        print(raw.decode(errors='replace')[-9000:], flush=True)
        raise SystemExit('Unexpected actual result; preserve and stop, no blind retry')
    return raw
