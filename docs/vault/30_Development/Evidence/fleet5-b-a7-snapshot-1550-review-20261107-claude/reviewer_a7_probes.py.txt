"""Reviewer A7 snapshot R1 probes: AST/source-only, serial, one process.

Usage: python -B -I reviewer_a7_probes.py <a7 evidence dir abs> <out.json>
Calls the owner's final a7_body.prove / a7_inventory.census on in-memory
mutated sources. Nothing from the product or a mutant is imported or executed;
every mutant is only compile()d for syntax.
"""
import ast, hashlib, json, sys, time
from collections import Counter
from pathlib import Path

E = Path(sys.argv[1]).resolve(); OUT = Path(sys.argv[2])
assert E.is_dir() and (E / 'a7_body.py').exists(), E
sys.path.insert(0, str(E))

# Precondition 1: every manifest-pinned file/producer equals the committed bytes.
manifest_bytes = (E / 'proof-producer-hashes.json').read_bytes()
manifest = json.loads(manifest_bytes)
mism = [r['file'] for r in manifest['files'] if hashlib.sha256((E / r['file']).read_bytes()).hexdigest() != r['sha256']]
pmism = [k for k, v in manifest['producers'].items()
         if hashlib.sha256(((E.parent / k[3:]) if k.startswith('../') else (E / k)).read_bytes()).hexdigest() != v]
pre = {'manifestSHA256': hashlib.sha256(manifest_bytes).hexdigest(), 'manifestFiles': len(manifest['files']), 'fileMismatches': mism,
       'producers': len(manifest['producers']), 'producerMismatches': pmism}
assert pre['manifestSHA256'] == '2a75a22f883f9ab9eba13e8b618c213710f7880f5df39769f10d8ab6bcf4ee43' and not mism and not pmism, pre

import a7_body as body, a7_inventory as inventory, use_sites as u, reflection_audit as ra
from a7_common import NC, CT, DB, ER, PROTECTED, frozen_source, sha
from source_bindings import Facts

rows = []
def log(row):
    rows.append(row); print(json.dumps(row, sort_keys=True)[:1500], flush=True)

def replace(path, old, new):
    s = frozen_source(path); assert s.count(old) == 1, (path, old, s.count(old)); return {path: s.replace(old, new, 1)}
def append(path, extra):
    return {path: frozen_source(path) + extra}

def injected_spans(path, world, marker):
    s = world[path]; t = ast.parse(s); out = []
    for n in ast.walk(t):
        if isinstance(n, ast.stmt) and marker in (ast.get_source_segment(s, n) or '') and not isinstance(n, (ast.FunctionDef, ast.ClassDef, ast.With, ast.If)):
            out.append({'span': [n.lineno, n.col_offset, n.end_lineno, n.end_col_offset], 'text': ast.get_source_segment(s, n)})
    return out

def precheck(world):
    for p, s in world.items(): compile(s, p, 'exec', dont_inherit=True)
    return {'inputSHA256': {p: sha(s) for p, s in world.items()}, 'frozenSHA256': {p: sha(frozen_source(p)) for p in world if not p.endswith('a7_reviewer_new_file.py')},
            'unchangedSelected': {p: sha(frozen_source(p)) for p in (NC, CT, DB, ER) if p not in world}}

def run_body(name, world, expect, note=''):
    pc = precheck(world); t0 = time.monotonic()
    with body.target_scope():
        r = body.prove(world)
    row = {'layer': 'BODY', 'case': name, 'accepted': r['bodyAccepted'], 'reason': r['reason'], 'detail': r.get('detail'), 'expected': expect,
           'gatesBeforeRefusal': [c['gate'] for c in r['checks']], 'seconds': round(time.monotonic() - t0, 1), 'note': note, **pc}
    row['matches'] = (r['bodyAccepted'] if expect == 'ACCEPT' else (not r['bodyAccepted'] and (expect == 'REFUSE' or expect in str(r['reason']))))
    log(row); return row

BASE = None
def obl_key(o):
    s = o.get('source') or {}
    return (o['code'], s.get('path') if isinstance(s, dict) else str(s), tuple(s.get('span', [])) if isinstance(s, dict) else ())

def run_census(name, world, note=''):
    pc = precheck(world); t0 = time.monotonic()
    try:
        with body.target_scope():
            c = inventory.census(world)
    except Exception as e:  # recorded as a harness/precondition crash, never as a refusal or a pass
        row = {'layer': 'CENSUS', 'case': name, 'exception': type(e).__name__ + ':' + str(e)[:300], 'note': note, **pc}; log(row); return row
    cur = Counter(obl_key(o) for o in c['obligations'])
    added = sorted([list(map(str, k)) + [n] for k, n in (cur - BASE['obl']).items()]); removed = sorted([list(map(str, k)) + [n] for k, n in (BASE['obl'] - cur).items()])
    files = {x['path']: x['retainedForCallClosure'] for x in c['sourceFiles']}
    row = {'layer': 'CENSUS', 'case': name, 'obligations': len(c['obligations']), 'addedVsBaseline': added, 'removedVsBaseline': removed,
           'constructors': len(c['constructorSites']), 'snapshots': len(c['snapshotSites']),
           'otherReceivers': [{'path': x['source']['path'], 'span': x['source']['span'], 'receiver': x['receiver'], 'purpose': x['purpose']} for x in c['otherSnapshotReceivers']],
           'retainedForCallClosure': {p: files.get(p) for p in world}, 'seconds': round(time.monotonic() - t0, 1), 'note': note, **pc}
    log(row); return row

def guard_only(name, path, source):
    with body.target_scope():
        f = Facts(path, source); a = ra.audit(f, PROTECTED); uses, error = u.module_uses(source, path, instances=True, factories=(), parameter_context={})
    row = {'layer': 'GUARD_ONLY', 'case': name, 'path': path, 'inputSHA256': sha(source), 'reflectionNewRelevantHits': a['newRelevantHits'], 'useSiteError': error}
    log(row); return row

# ---- baseline (actual 72028) ----
with body.target_scope():
    c0 = inventory.census(None)
BASE = {'obl': Counter(obl_key(o) for o in c0['obligations'])}
log({'layer': 'CENSUS', 'case': 'baseline-actual-72028', 'censusObligations': len(c0['obligations']), 'constructors': len(c0['constructorSites']), 'snapshots': len(c0['snapshotSites']),
     'other': len(c0['otherSnapshotReceivers']), 'parsed': len(c0['sourceFiles']), 'retained': sum(x['retainedForCallClosure'] for x in c0['sourceFiles']), 'pre': pre})

ret = OBS = 'services/control-plane/src/inv/observation.py'
